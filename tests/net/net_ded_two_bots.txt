# The dedicated server with two remote players and bots, Player Collision Off: csx-server on de_aztec with 4 bots
# (normal difficulty) and every player slot for remote players and bots (no host player). Two clients join, one per
# team; the server restarts the round from its console twice (sv_restartround: round ends, respawns); the bots
# fight. Each client moves, opens the scoreboard (6 players: 2 humans with pings, 4 BOT rows) and stays connected.
# Shows: humans and bots together on a server of its own, rounds, Collision Off reaching the clients, the scoreboard.
# timeout: 600
[server] --server "--map de_aztec --bots 4 --bot-difficulty 1 --player-collision off" --env CSX_LOG_NET=1
wait log net:.server.identity
count net:.client.127\.0\.0\.[23]:[0-9]+.connected 2
count joined.team 6
sleep 3
pckey cmd sv_restartround 1
sleep 8
players r1
expect playercount r1 6 6
expect count playerping.r1.[0-9]+.[0-9]+$ 2
pckey cmd sv_restartround 1
sleep 8
count World.triggered."Round_End" 2
players r2
expect playercount r2 6 6
expect log net.stats:.snapshots.a.second.per.client:.(29|30)\.[0-9].mean
expect nolog guest trap
expect nolog Host_Error
expect nolog net:.client.*closed
sleep 20
pckey cmd quit
[client*2] --after server:net:.server.identity --delay 3 --stagger 3 --launch tests/launch/connect_local --env FPS=30 --env CSX_LOG_NET=1 --env CSX_LOG_SCOREBOARD=1
wait log net:.server.127\.0\.0\.1:27015.connected
wait log pc:.player.collision.off
wait log pc.keys:.menu.TeamSelect
sleep 2
press A
sleep 4
press B
sleep 25
state m1
pckey tap +moveleft 0.5
sleep 1.5
state m2
expect moved m1 m2 20 400
pckey cmd +showscores
sleep 2
expect log scoreboard:.6.players
pckey cmd -showscores
sleep 2
expect log scoreboard:.*ping=[0-9]+.human
expect nolog Host_Error
expect nolog guest trap
wait log (?i)net:.server.127\.0\.0\.1:27015.closed:.server.shut
