# An admin changes the map and the bot count of the dedicated server: admins.cfg gives the test key every
# permission. The admin sets "admin bots 3" (from the next map) and runs "admin map de_dust2": the server tells its
# players (MapChange) and relaunches into de_dust2 with 3 bots (the setting carried in its environment), the admin
# rejoins by itself; then the server quits.
# timeout: 420
[server] --server "--map de_aztec --bots 1" --file admins.cfg=tests/net/files/admins_all.cfg --env CSX_LOG_NET=1
wait log net:.server.identity
wait log net:.client.127\.0\.0\.2:[0-9]+.connected
wait log admin:.*ran."admin.bots.3":.bots.3.from.the.next.map
wait log admin:.*ran."admin.map.de_dust2":.map.change.to.de_dust2
wait log net:.relaunching.into.de_dust2:.the.clients
wait log server:.relaunched.on.UDP.port.27015,.map.de_dust2.*3.bots
wait log open.D:\\de_dust2\.sxwad
wait log net:.client.127\.0\.0\.2:[0-9]+.connected
count joined.team 4
sleep 5
players newmap
expect playercount newmap 4 4
expect nolog guest trap
expect nolog Host_Error
pckey cmd quit
[admin] --after server:net:.server.identity --delay 3 --file identity.key=tests/net/files/admin_identity.key --launch tests/launch/connect_local --env FPS=30 --env CSX_LOG_NET=1
wait log net:.server.127\.0\.0\.1:27015.connected
wait log pc.keys:.menu.TeamSelect
sleep 2
press A
sleep 4
press B
sleep 3
pckey cmd admin bots 3
sleep 1
pckey cmd admin map de_dust2
wait log net:.relaunching.to.rejoin
wait log net:.server.127\.0\.0\.1:27015.has.its.remembered.key
wait log net:.server.127\.0\.0\.1:27015.connected
wait log pc.keys:.menu.TeamSelect
sleep 2
press A
sleep 4
press B
sleep 10
expect nolog guest trap
wait log (?i)net:.server.127\.0\.0\.1:27015.closed:.server.shut
