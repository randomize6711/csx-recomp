# Joining from the server browser (docs/pc_multiplayer.md "N3"): Alpha (27015, 2 bots), Bravo (27016, password
# "letmein") and Full (27017, 4 slots, 4 bots) on the LAN list, sorted by name. A player joins Alpha (A: the title
# relaunches into the server, its key pinned); one joins Bravo with the password typed on the keyboard screen; one
# types a wrong password: the server refuses it, the game goes back to the front end and the browser shows why; one
# tries Full: "The server is full" before anything is sent. When Alpha shuts down, its player is back in the browser
# with "Disconnected: Server shutting down". The password is in no log.
# timeout: 480
[server] --server "--map de_aztec --bots 2 --name Alpha" --env CSX_LOG_NET=1
wait log net:.answering.server.queries
wait log net:.client.127\.0\.0\.4:[0-9]+.connected
sleep 120
pckey cmd quit
[pwserver] --server "--map cs_office --bots 0 --port 27016 --name Bravo --password letmein" --env CSX_LOG_NET=1
wait log net:.answering.server.queries
wait log net:.client.127\.0\.0\.5:[0-9]+.connected
wait log refused:.bad.password
sleep 60
expect nolog letmein
pckey cmd quit
[full] --server "--map de_dust2 --bots 4 --max-players 4 --port 27017 --name Full" --env CSX_LOG_NET=1
wait log net:.answering.server.queries
sleep 290

pckey cmd quit
