# Three clients join at the same moment (the same --after condition): each asks for a challenge in the same server
# frame. Regression for net_disconnect's failure (one of them was refused "no challenge" and stuck on LostHost).
# Shows: simultaneous joins, every client signs on and enters the game.
# timeout: 300
[host] --launch tests/launch/de_aztec_2humans --env FPS=30 --env CSX_LOG_CONSOLE=1
wait log open D:\\(de|cs)_[a-z0-9_]*\.sxwad
wait draws>=150
sleep 5
press A
sleep 4
press B
count net:.client.127\.0\.0\.[0-9]:[0-9]+.connected 3
count entered.the.game 4
sleep 20
expect nolog refused
expect nolog guest trap
[client*3] --after host:Round_Start --launch tests/launch/connect_local --env FPS=30 --env CSX_LOG_CONSOLE=1
wait log net:.server.127\.0\.0\.1:27015.connected
wait log pc.keys:.menu.TeamSelect
sleep 2
press A
sleep 4
press B
sleep 10
expect nolog LostHost
